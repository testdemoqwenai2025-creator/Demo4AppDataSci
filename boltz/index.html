38class MSAEncoder(nn.Module):
39    """MSA encoder for Boltz-1 (similar to AlphaFold2 Evoformer input).
40    
41    Input: MSA (B, M, L, vocab) — M sequences × L residues
42    Output: per-residue features (B, L, hidden_dim)
43    
44    Production: AlphaFold2 uses Evoformer (single + pair tracks).
45    Boltz-1 simplifies with transformer encoder.
46    """
47    def __init__(self, vocab_size: int = 25, hidden_dim: int = 256,
48                 num_layers: int = 4, num_heads: int = 8):
49        super().__init__()
50        self.token_embed = nn.Embedding(vocab_size, hidden_dim)
51        self.pos_embed = nn.Parameter(torch.zeros(1, 1024, hidden_dim))
52        encoder_layer = nn.TransformerEncoderLayer(
53            d_model=hidden_dim, nhead=num_heads, batch_first=True,
54            dim_feedforward=4 * hidden_dim, activation='gelu',
55            norm_first=True,
56        )
57        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers)
58        self.norm = nn.LayerNorm(hidden_dim)
59    
60    def forward(self, msa_tokens: torch.Tensor) -> torch.Tensor:
61        """Encode MSA → per-residue features.
62        
63        Args:
64            msa_tokens: (B, M, L) MSA tokens (M sequences × L residues)
65        
66        Returns: (B, L, hidden_dim) per-residue features
67        """
68        B, M, L = msa_tokens.shape
69        # Embed tokens
70        x = self.token_embed(msa_tokens)  # (B, M, L, hidden)
71        # Add positional embedding
72        x = x + self.pos_embed[:, :L]
73        # Reshape to (B*M, L, hidden) for transformer
74        x = x.reshape(B * M, L, -1)
75        x = self.transformer(x)
76        x = self.norm(x)
77        # Reshape back and pool over MSA dimension
78        x = x.reshape(B, M, L, -1).mean(dim=1)  # (B, L, hidden)
79        return x

87

95        # Outer product mean: pair features from single rep
96        self.outer_proj = nn.Linear(hidden_dim, hidden_dim // 2)
97    
98    def forward(self, single_rep: torch.Tensor) -> torch.Tensor:
99        """Compute pair representation.
100        
101        Args:
102            single_rep: (B, L, hidden_dim) per-residue features
103        
104        Returns: (B, L, L, hidden_dim) pair features
105        """
106        B, L, H = single_rep.shape
107        # Outer product: pair[i, j] = outer(single[i], single[j])
108        proj = self.outer_proj(single_rep)  # (B, L, H/2)
109        # Outer product mean
110        outer = torch.einsum('bid,bjc->bijd', proj, proj)  # (B, L, L, H/2 * H/2)
111        # Project back to hidden_dim
112        return outer.reshape(B, L, L, -1)

134        # Time embedding
135        self.time_embed = nn.Sequential(
136            nn.Linear(1, hidden_dim),
137            nn.SiLU(),
138            nn.Linear(hidden_dim, hidden_dim),
139        )
140        
141        # SE(3)-equivariant denoising layers (simplified IPA)
142        self.layers = nn.ModuleList([
143            IPALayer(hidden_dim, n_heads) for _ in range(num_layers)
144        ])
145        
146        # Noise prediction head (per-atom 3D vector)
147        self.noise_head = nn.Linear(hidden_dim, 3)
148        
149        # Noise schedule (cosine, like ADR-027/036)
150        betas = torch.linspace(1e-4, 0.02, n_diffusion_steps)
151        alphas = 1 - betas
152        alpha_bars = torch.cumprod(alphas, dim=0)
153        self.register_buffer('betas', betas)
154        self.register_buffer('alphas', alphas)
155        self.register_buffer('alpha_bars', alpha_bars)
156    
157    def forward(self, atom_coords: torch.Tensor, atom_emb: torch.Tensor,
158                msa_features: torch.Tensor, pair_features: torch.Tensor,
159                t: int) -> torch.Tensor:
160        """Predict noise ε_theta(x_t, t, condition).
161        
162        Args:
163            atom_coords: (B, N, 3) noisy coordinates
164            atom_emb: (B, N, hidden_dim) atom type embeddings
165            msa_features: (B, N, hidden_dim) per-residue MSA features
166            pair_features: (B, N, N, hidden_dim) pair features
167            t: int diffusion timestep
168        
169        Returns: (B, N, 3) predicted noise
170        """
171        B, N, _ = atom_coords.shape
172        
173        # Combine features
174        h = atom_emb + msa_features  # (B, N, hidden)
175        
176        # Time embedding
177        t_emb = self.time_embed(torch.tensor([t], device=atom_coords.device).float().unsqueeze(0))  # (1, hidden)
178        h = h + t_emb.unsqueeze(1)  # broadcast over atoms
179        
180        # SE(3)-equivariant denoising layers
181        for layer in self.layers:
182            h, atom_coords = layer(h, atom_coords, pair_features)
183        
184        # Predict noise (per-atom 3D vector)
185        noise = self.noise_head(h)  # (B, N, 3)
186        return noise
187    
188    @torch.no_grad()
189    def sample(self, n_atoms: int, atom_emb: torch.Tensor,
190               msa_features: torch.Tensor, pair_features: torch.Tensor,
191               save_intermediate: bool = False) -> torch.Tensor:
192        """Reverse diffusion: from noise to structure.
193        
194        Args:
195            n_atoms: number of atoms to predict
196            atom_emb: (B, n_atoms, hidden) atom type embeddings (conditioning)
197            msa_features: per-residue MSA features
198            pair_features: pair features
199        
200        Returns: (B, n_atoms, 3) predicted 3D coordinates
201        """
202        B = atom_emb.shape[0]
203        
204        # Initialise from noise
205        x = torch.randn(B, n_atoms, 3, device=atom_emb.device)
206        
207        intermediate = [x.clone()]
208        
209        # Reverse diffusion
210        for t in range(self.n_diffusion_steps - 1, -1, -1):
211            noise = self.forward(x, atom_emb, msa_features, pair_features, t)
212            
213            alpha_t = self.alphas[t]
214            alpha_bar_t = self.alpha_bars[t]
215            beta_t = self.betas[t]
216            
217            # DDPM reverse update
218            x_mean = (x - (beta_t / torch.sqrt(1 - alpha_bar_t)) * noise) / torch.sqrt(alpha_t)
219            if t > 0:
220                sigma_t = torch.sqrt(beta_t * (1 - alpha_bar_t) / (1 - alpha_bar_t))
221                x = x_mean + sigma_t * torch.randn_like(x)
222            else:
223                x = x_mean
224            
225            if save_intermediate:
226                intermediate.append(x.clone())
227        
228        if save_intermediate:
229            return x, torch.stack(intermediate)
230        return x

232class IPALayer(nn.Module):
233    """Invariant Point Attention layer (AlphaFold2/Boltz-1).
234    
235    SE(3)-equivariant: f(Rx) = Rf(x).
236    
237    Combines standard self-attention with 3D geometric attention.
238    """
239    def __init__(self, hidden_dim: int = 256, n_heads: int = 8):
240        super().__init__()
241        self.n_heads = n_heads
242        # Standard attention
243        self.q_proj = nn.Linear(hidden_dim, hidden_dim)
244        self.k_proj = nn.Linear(hidden_dim, hidden_dim)
245        self.v_proj = nn.Linear(hidden_dim, hidden_dim)
246        # 3D point attention
247        self.q_point = nn.Linear(hidden_dim, n_heads * 3)
248        self.k_point = nn.Linear(hidden_dim, n_heads * 3)
249        # Output projection
250        self.out_proj = nn.Linear(hidden_dim * 2, hidden_dim)
251        self.norm = nn.LayerNorm(hidden_dim)
252    
253    def forward(self, h: torch.Tensor, coords: torch.Tensor,
254                pair_features: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
255        """Forward pass.
256        
257        Args:
258            h: (B, N, hidden) atom features
259            coords: (B, N, 3) atom coordinates
260            pair_features: (B, N, N, hidden) pair features
261        
262        Returns: updated (h, coords)
263        """
264        B, N, _ = h.shape
265        H = self.n_heads
266        
267        # Standard attention
268        q = self.q_proj(h).view(B, N, H, -1)
269        k = self.k_proj(h).view(B, N, H, -1)
270        v = self.v_proj(h).view(B, N, H, -1)
271        
272        # Attention scores
273        attn = (q @ k.transpose(-2, -1)) / math.sqrt(H)
274        attn = F.softmax(attn, dim=-1)
275        
276        # Standard attention output
277        std_out = (attn @ v).reshape(B, N, -1)
278        
279        # 3D point attention (IPA)
280        q_pt = self.q_point(h).view(B, N, H, 3)
281        k_pt = self.k_point(h).view(B, N, H, 3)
282        
283        # 3D distance attention
284        dist = (q_pt.unsqueeze(2) - k_pt.unsqueeze(1)).norm(dim=-1)  # (B, N, N, H)
285        pt_attn = torch.exp(-dist / 10.0)  # Gaussian falloff
286        pt_attn = pt_attn / (pt_attn.sum(dim=2, keepdim=True) + 1e-8)
287        
288        # Combined output
289        combined = torch.cat([std_out, std_out[:, :, :H]], dim=-1)  # simplified
290        h = self.norm(h + self.out_proj(combined))
291        
292        # Update coords (small step — IPA is equivariant by construction)
293        delta_x = 0.1 * (q_pt * pt_attn.sum(dim=1).unsqueeze(-1)).sum(dim=1, keepdim=True)
294        coords = coords + delta_x.squeeze(1) * 0.1  # tiny update
295        
296        return h, coords

302class ConfidenceHead(nn.Module):
303    """Per-atom pLDDT + per-pair interface ipTM.
304    
305    Production: AlphaFold2-style confidence head.
306    """
307    def __init__(self, hidden_dim: int = 256):
308        super().__init__()
309        self.plddt_head = nn.Sequential(
310            nn.Linear(hidden_dim, hidden_dim // 2),
311            nn.ReLU(),
312            nn.Linear(hidden_dim // 2, 1),
313            nn.Sigmoid(),
314        )
315        # Pairwise contact prediction
316        self.contact_head = nn.Sequential(
317            nn.Linear(hidden_dim * 2, hidden_dim),
318            nn.ReLU(),
319            nn.Linear(hidden_dim, 1),
320            nn.Sigmoid(),
321        )
322    
323    def forward(self, h: torch.Tensor) -> Dict[str, torch.Tensor]:
324        """Predict per-atom pLDDT + pairwise contact.
325        
326        Args:
327            h: (B, N, hidden) atom features
328        
329        Returns: dict with 'plddt' (B, N), 'contact' (B, N, N)
330        """
331        plddt = self.plddt_head(h).squeeze(-1)  # (B, N)
332        
333        # Pairwise contact via outer product
334        h_i = h.unsqueeze(2)  # (B, N, 1, hidden)
335        h_j = h.unsqueeze(1)  # (B, 1, N, hidden)
336        pair_h = torch.cat([h_i.expand(-1, -1, h.shape[1], -1),
337                            h_j.expand(-1, h.shape[1], -1, -1)], dim=-1)
338        contact = self.contact_head(pair_h).squeeze(-1)  # (B, N, N)
339        
340        return {'plddt': plddt, 'contact': contact}

348

350        1. Atom type embedding (mixed atoms)
351        2. MSA encoder (per-residue features)
352        3. Pair representation (residue-residue)
353        4. Diffusion structure module (SE(3)-equivariant)
354        5. Confidence head (pLDDT + ipTM)

358    """
359    def __init__(self, n_atom_types: int = len(ATOM_TYPES),
360                 hidden_dim: int = 256, msa_layers: int = 4,
361                 struct_layers: int = 8, n_diffusion_steps: int = 50):
362        super().__init__()
363        self.atom_embedding = AtomTypeEmbedding(n_atom_types, hidden_dim)
364        self.msa_encoder = MSAEncoder(vocab_size=25, hidden_dim=hidden_dim, num_layers=msa_layers)
365        self.pair_rep = PairRepresentation(hidden_dim)
366        self.structure_module = BoltzStructureModule(
367            hidden_dim, struct_layers, n_diffusion_steps=n_diffusion_steps
368        )
369        self.confidence_head = ConfidenceHead(hidden_dim)
370    
371    def forward(self, atom_types: torch.Tensor, msa_tokens: torch.Tensor,
372                save_intermediate: bool = False) -> Dict[str, torch.Tensor]:
373        """Predict structure + confidence.
374        
375        Args:
376            atom_types: (B, N) integer atom type IDs
377            msa_tokens: (B, M, L) MSA tokens (M sequences × L residues)
378        
379        Returns: dict with 'coords', 'plddt', 'contact'
380        """
381        # Encode
382        atom_emb = self.atom_embedding(atom_types)  # (B, N, hidden)
383        msa_features = self.msa_encoder(msa_tokens)  # (B, L, hidden)
384        # Broadcast msa_features to atom count
385        N = atom_types.shape[1]
386        if msa_features.shape[1] != N:
387            # Interpolate (simplified — production handles residue-to-atom mapping)
388            msa_features = F.interpolate(msa_features.transpose(1, 2), size=N, mode='linear').transpose(1, 2)
389        pair_features = self.pair_rep(msa_features)  # (B, L, L, hidden)
390        
391        # Sample structure (reverse diffusion)
392        if save_intermediate:
393            coords, intermediate = self.structure_module.sample(
394                N, atom_emb, msa_features, pair_features, save_intermediate=True
395            )
396        else:
397            coords = self.structure_module.sample(N, atom_emb, msa_features, pair_features)
398        
399        # Predict confidence
400        # Run structure module forward for confidence (using final coords)
401        h_final = atom_emb + msa_features  # simplified
402        confidence = self.confidence_head(h_final)
403        
404        out = {
405            'coords': coords,
406            'plddt': confidence['plddt'],
407            'contact': confidence['contact'],
408        }
409        if save_intermediate:
410            out['intermediate'] = intermediate
411        return out

38class MSAEncoder(nn.Module):
39    """MSA encoder for Boltz-1 (similar to AlphaFold2 Evoformer input).
40    
41    Input: MSA (B, M, L, vocab) — M sequences × L residues
42    Output: per-residue features (B, L, hidden_dim)
43    
44    Production: AlphaFold2 uses Evoformer (single + pair tracks).
45    Boltz-1 simplifies with transformer encoder.
46    """
47    def __init__(self, vocab_size: int = 25, hidden_dim: int = 256,
48                 num_layers: int = 4, num_heads: int = 8):
49        super().__init__()
50        self.token_embed = nn.Embedding(vocab_size, hidden_dim)
51        self.pos_embed = nn.Parameter(torch.zeros(1, 1024, hidden_dim))
52        encoder_layer = nn.TransformerEncoderLayer(
53            d_model=hidden_dim, nhead=num_heads, batch_first=True,
54            dim_feedforward=4 * hidden_dim, activation='gelu',
55            norm_first=True,
56        )
57        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers)
58        self.norm = nn.LayerNorm(hidden_dim)
59    
60    def forward(self, msa_tokens: torch.Tensor) -> torch.Tensor:
61        """Encode MSA → per-residue features.
62        
63        Args:
64            msa_tokens: (B, M, L) MSA tokens (M sequences × L residues)
65        
66        Returns: (B, L, hidden_dim) per-residue features
67        """
68        B, M, L = msa_tokens.shape
69        # Embed tokens
70        x = self.token_embed(msa_tokens)  # (B, M, L, hidden)
71        # Add positional embedding
72        x = x + self.pos_embed[:, :L]
73        # Reshape to (B*M, L, hidden) for transformer
74        x = x.reshape(B * M, L, -1)
75        x = self.transformer(x)
76        x = self.norm(x)
77        # Reshape back and pool over MSA dimension
78        x = x.reshape(B, M, L, -1).mean(dim=1)  # (B, L, hidden)
79        return x

85class PairRepresentation(nn.Module):
86    """Pair representation (Boltz-1 / AlphaFold2 Evoformer).
87    
88    For each residue pair (i, j): relative position, contact prediction.
89    Used by diffusion module to model inter-residue distances.
90    """
91    def __init__(self, hidden_dim: int = 256):
92        super().__init__()
93        # Project relative position to pair features
94        self.rel_pos_proj = nn.Linear(128, hidden_dim)  # sinusoidal rel pos
95        # Outer product mean: pair features from single rep
96        self.outer_proj = nn.Linear(hidden_dim, hidden_dim // 2)
97    
98    def forward(self, single_rep: torch.Tensor) -> torch.Tensor:
99        """Compute pair representation.
100        
101        Args:
102            single_rep: (B, L, hidden_dim) per-residue features
103        
104        Returns: (B, L, L, hidden_dim) pair features
105        """
106        B, L, H = single_rep.shape
107        # Outer product: pair[i, j] = outer(single[i], single[j])
108        proj = self.outer_proj(single_rep)  # (B, L, H/2)
109        # Outer product mean
110        outer = torch.einsum('bid,bjc->bijd', proj, proj)  # (B, L, L, H/2 * H/2)
111        # Project back to hidden_dim
112        return outer.reshape(B, L, L, -1)

118class BoltzStructureModule(nn.Module):
119    """Boltz-1 structure module (AlphaFold3-style diffusion).
120    
121    Architecture:
122        - Diffusion over (N × 3) atom coordinates
123        - SE(3)-equivariant denoising (IPA-style attention)
124        - Atom-type-conditioned (different atom types have different denoising)
125    
126    Production: same as AlphaFold3 — DiffusionTransformer (DiT) + IPA.
127    """
128    def __init__(self, hidden_dim: int = 256, num_layers: int = 8,
129                 n_heads: int = 8, n_diffusion_steps: int = 50):
130        super().__init__()
131        self.n_diffusion_steps = n_diffusion_steps
132        self.hidden_dim = hidden_dim
133        
134        # Time embedding
135        self.time_embed = nn.Sequential(
136            nn.Linear(1, hidden_dim),
137            nn.SiLU(),
138            nn.Linear(hidden_dim, hidden_dim),
139        )
140        
141        # SE(3)-equivariant denoising layers (simplified IPA)
142        self.layers = nn.ModuleList([
143            IPALayer(hidden_dim, n_heads) for _ in range(num_layers)
144        ])
145        
146        # Noise prediction head (per-atom 3D vector)
147        self.noise_head = nn.Linear(hidden_dim, 3)
148        
149        # Noise schedule (cosine, like ADR-027/036)
150        betas = torch.linspace(1e-4, 0.02, n_diffusion_steps)
151        alphas = 1 - betas
152        alpha_bars = torch.cumprod(alphas, dim=0)
153        self.register_buffer('betas', betas)
154        self.register_buffer('alphas', alphas)
155        self.register_buffer('alpha_bars', alpha_bars)
156    
157    def forward(self, atom_coords: torch.Tensor, atom_emb: torch.Tensor,
158                msa_features: torch.Tensor, pair_features: torch.Tensor,
159                t: int) -> torch.Tensor:
160        """Predict noise ε_theta(x_t, t, condition).
161        
162        Args:
163            atom_coords: (B, N, 3) noisy coordinates
164            atom_emb: (B, N, hidden_dim) atom type embeddings
165            msa_features: (B, N, hidden_dim) per-residue MSA features
166            pair_features: (B, N, N, hidden_dim) pair features
167            t: int diffusion timestep
168        
169        Returns: (B, N, 3) predicted noise
170        """
171        B, N, _ = atom_coords.shape
172        
173        # Combine features
174        h = atom_emb + msa_features  # (B, N, hidden)
175        
176        # Time embedding
177        t_emb = self.time_embed(torch.tensor([t], device=atom_coords.device).float().unsqueeze(0))  # (1, hidden)
178        h = h + t_emb.unsqueeze(1)  # broadcast over atoms
179        
180        # SE(3)-equivariant denoising layers
181        for layer in self.layers:
182            h, atom_coords = layer(h, atom_coords, pair_features)
183        
184        # Predict noise (per-atom 3D vector)
185        noise = self.noise_head(h)  # (B, N, 3)
186        return noise
187    
188    @torch.no_grad()
189    def sample(self, n_atoms: int, atom_emb: torch.Tensor,
190               msa_features: torch.Tensor, pair_features: torch.Tensor,
191               save_intermediate: bool = False) -> torch.Tensor:
192        """Reverse diffusion: from noise to structure.
193        
194        Args:
195            n_atoms: number of atoms to predict
196            atom_emb: (B, n_atoms, hidden) atom type embeddings (conditioning)
197            msa_features: per-residue MSA features
198            pair_features: pair features
199        
200        Returns: (B, n_atoms, 3) predicted 3D coordinates
201        """
202        B = atom_emb.shape[0]
203        
204        # Initialise from noise
205        x = torch.randn(B, n_atoms, 3, device=atom_emb.device)
206        
207        intermediate = [x.clone()]
208        
209        # Reverse diffusion
210        for t in range(self.n_diffusion_steps - 1, -1, -1):
211            noise = self.forward(x, atom_emb, msa_features, pair_features, t)
212            
213            alpha_t = self.alphas[t]
214            alpha_bar_t = self.alpha_bars[t]
215            beta_t = self.betas[t]
216            
217            # DDPM reverse update
218            x_mean = (x - (beta_t / torch.sqrt(1 - alpha_bar_t)) * noise) / torch.sqrt(alpha_t)
219            if t > 0:
220                sigma_t = torch.sqrt(beta_t * (1 - alpha_bar_t) / (1 - alpha_bar_t))
221                x = x_mean + sigma_t * torch.randn_like(x)
222            else:
223                x = x_mean
224            
225            if save_intermediate:
226                intermediate.append(x.clone())
227        
228        if save_intermediate:
229            return x, torch.stack(intermediate)
230        return x

232class IPALayer(nn.Module):
233    """Invariant Point Attention layer (AlphaFold2/Boltz-1).
234    
235    SE(3)-equivariant: f(Rx) = Rf(x).
236    
237    Combines standard self-attention with 3D geometric attention.
238    """
239    def __init__(self, hidden_dim: int = 256, n_heads: int = 8):
240        super().__init__()
241        self.n_heads = n_heads
242        # Standard attention
243        self.q_proj = nn.Linear(hidden_dim, hidden_dim)
244        self.k_proj = nn.Linear(hidden_dim, hidden_dim)
245        self.v_proj = nn.Linear(hidden_dim, hidden_dim)
246        # 3D point attention
247        self.q_point = nn.Linear(hidden_dim, n_heads * 3)
248        self.k_point = nn.Linear(hidden_dim, n_heads * 3)
249        # Output projection
250        self.out_proj = nn.Linear(hidden_dim * 2, hidden_dim)
251        self.norm = nn.LayerNorm(hidden_dim)
252    
253    def forward(self, h: torch.Tensor, coords: torch.Tensor,
254                pair_features: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
255        """Forward pass.
256        
257        Args:
258            h: (B, N, hidden) atom features
259            coords: (B, N, 3) atom coordinates
260            pair_features: (B, N, N, hidden) pair features
261        
262        Returns: updated (h, coords)
263        """
264        B, N, _ = h.shape
265        H = self.n_heads
266        
267        # Standard attention
268        q = self.q_proj(h).view(B, N, H, -1)
269        k = self.k_proj(h).view(B, N, H, -1)
270        v = self.v_proj(h).view(B, N, H, -1)
271        
272        # Attention scores
273        attn = (q @ k.transpose(-2, -1)) / math.sqrt(H)
274        attn = F.softmax(attn, dim=-1)
275        
276        # Standard attention output
277        std_out = (attn @ v).reshape(B, N, -1)
278        
279        # 3D point attention (IPA)
280        q_pt = self.q_point(h).view(B, N, H, 3)
281        k_pt = self.k_point(h).view(B, N, H, 3)
282        
283        # 3D distance attention
284        dist = (q_pt.unsqueeze(2) - k_pt.unsqueeze(1)).norm(dim=-1)  # (B, N, N, H)
285        pt_attn = torch.exp(-dist / 10.0)  # Gaussian falloff
286        pt_attn = pt_attn / (pt_attn.sum(dim=2, keepdim=True) + 1e-8)
287        
288        # Combined output
289        combined = torch.cat([std_out, std_out[:, :, :H]], dim=-1)  # simplified
290        h = self.norm(h + self.out_proj(combined))
291        
292        # Update coords (small step — IPA is equivariant by construction)
293        delta_x = 0.1 * (q_pt * pt_attn.sum(dim=1).unsqueeze(-1)).sum(dim=1, keepdim=True)
294        coords = coords + delta_x.squeeze(1) * 0.1  # tiny update
295        
296        return h, coords

302class ConfidenceHead(nn.Module):
303    """Per-atom pLDDT + per-pair interface ipTM.
304    
305    Production: AlphaFold2-style confidence head.
306    """
307    def __init__(self, hidden_dim: int = 256):
308        super().__init__()
309        self.plddt_head = nn.Sequential(
310            nn.Linear(hidden_dim, hidden_dim // 2),
311            nn.ReLU(),
312            nn.Linear(hidden_dim // 2, 1),
313            nn.Sigmoid(),
314        )
315        # Pairwise contact prediction
316        self.contact_head = nn.Sequential(
317            nn.Linear(hidden_dim * 2, hidden_dim),
318            nn.ReLU(),
319            nn.Linear(hidden_dim, 1),
320            nn.Sigmoid(),
321        )
322    
323    def forward(self, h: torch.Tensor) -> Dict[str, torch.Tensor]:
324        """Predict per-atom pLDDT + pairwise contact.
325        
326        Args:
327            h: (B, N, hidden) atom features
328        
329        Returns: dict with 'plddt' (B, N), 'contact' (B, N, N)
330        """
331        plddt = self.plddt_head(h).squeeze(-1)  # (B, N)
332        
333        # Pairwise contact via outer product
334        h_i = h.unsqueeze(2)  # (B, N, 1, hidden)
335        h_j = h.unsqueeze(1)  # (B, 1, N, hidden)
336        pair_h = torch.cat([h_i.expand(-1, -1, h.shape[1], -1),
337                            h_j.expand(-1, h.shape[1], -1, -1)], dim=-1)
338        contact = self.contact_head(pair_h).squeeze(-1)  # (B, N, N)
339        
340        return {'plddt': plddt, 'contact': contact}

346class Boltz1(nn.Module):
347    """Full Boltz-1 model (open-source AlphaFold3).
348    
349    Architecture:
350        1. Atom type embedding (mixed atoms)
351        2. MSA encoder (per-residue features)
352        3. Pair representation (residue-residue)
353        4. Diffusion structure module (SE(3)-equivariant)
354        5. Confidence head (pLDDT + ipTM)
355    
356    Training: same as AlphaFold3 (Abramson 2024).
357    Production: 773M params, ~10 min per complex on 1× A100.
358    """
359    def __init__(self, n_atom_types: int = len(ATOM_TYPES),
360                 hidden_dim: int = 256, msa_layers: int = 4,
361                 struct_layers: int = 8, n_diffusion_steps: int = 50):
362        super().__init__()
363        self.atom_embedding = AtomTypeEmbedding(n_atom_types, hidden_dim)
364        self.msa_encoder = MSAEncoder(vocab_size=25, hidden_dim=hidden_dim, num_layers=msa_layers)
365        self.pair_rep = PairRepresentation(hidden_dim)
366        self.structure_module = BoltzStructureModule(
367            hidden_dim, struct_layers, n_diffusion_steps=n_diffusion_steps
368        )
369        self.confidence_head = ConfidenceHead(hidden_dim)
370    
371    def forward(self, atom_types: torch.Tensor, msa_tokens: torch.Tensor,
372                save_intermediate: bool = False) -> Dict[str, torch.Tensor]:
373        """Predict structure + confidence.
374        
375        Args:
376            atom_types: (B, N) integer atom type IDs
377            msa_tokens: (B, M, L) MSA tokens (M sequences × L residues)
378        
379        Returns: dict with 'coords', 'plddt', 'contact'
380        """
381        # Encode
382        atom_emb = self.atom_embedding(atom_types)  # (B, N, hidden)
383        msa_features = self.msa_encoder(msa_tokens)  # (B, L, hidden)
384        # Broadcast msa_features to atom count
385        N = atom_types.shape[1]
386        if msa_features.shape[1] != N:
387            # Interpolate (simplified — production handles residue-to-atom mapping)
388            msa_features = F.interpolate(msa_features.transpose(1, 2), size=N, mode='linear').transpose(1, 2)
389        pair_features = self.pair_rep(msa_features)  # (B, L, L, hidden)
390        
391        # Sample structure (reverse diffusion)
392        if save_intermediate:
393            coords, intermediate = self.structure_module.sample(
394                N, atom_emb, msa_features, pair_features, save_intermediate=True
395            )
396        else:
397            coords = self.structure_module.sample(N, atom_emb, msa_features, pair_features)
398        
399        # Predict confidence
400        # Run structure module forward for confidence (using final coords)
401        h_final = atom_emb + msa_features  # simplified
402        confidence = self.confidence_head(h_final)
403        
404        out = {
405            'coords': coords,
406            'plddt': confidence['plddt'],
407            'contact': confidence['contact'],
408        }
409        if save_intermediate:
410            out['intermediate'] = intermediate
411        return out
